let roundQuestions = [];

let currentQIndex = 0;

let currentCorrectAnswer = null;


document.getElementById('start-quiz-btn').onclick = () => {

  roomView.classList.remove('active');

  quizView.classList.add('active');

  startRound();

function startRound() {

  // ランダムに5問選ぶ

  const shuffled = [...wordData].sort(() => 0.5 - Math.random());

  roundQuestions = shuffled.slice(0, 5);

  currentQIndex = 0;

  showNextQuestion();

}


function showNextQuestion() {

  if (currentQIndex >= roundQuestions.length) {

    // ラウンド終了

    quizView.classList.remove('active');

    roomView.classList.add('active');

    initRoom();

    return;

  }

  

  document.getElementById('quiz-progress-text').innerText = `だい ${currentQIndex + 1} もん！ (${currentQIndex + 1}/5)`;

  

  const qData = roundQuestions[currentQIndex];

  currentCorrectAnswer = qData;

  const contentEl = document.getElementById('quiz-content');

  contentEl.innerHTML = "";

  

  // クイズ形式をランダムに決定 (1: 絵+音声選択, 2: 発音, 3: スペル選択)

  // ※ 音声認識が未対応のブラウザでは 1 か 3 にする

  let qType = Math.floor(Math.random() * 3) + 1;

  if (!recognition && qType === 2) qType = 1;

  

  if (qType === 1) {

    // 形式1：音声を聞いて正しい絵を選ぶ

    contentEl.innerHTML = `

      <p style="font-size:1.5rem; font-weight:bold;">おとをきいて、えらぼう！</p>

      <button id="play-voice-btn">🔊</button>

      <div class="options-grid" id="options-container"></div>

    `;

    document.getElementById('play-voice-btn').onclick = () => playVoice(qData.en);

    createOptions(qData, contentEl.querySelector('#options-container'), true);

    setTimeout(() => playVoice(qData.en), 500);

    

  } else if (qType === 2) {

    // 形式2：マイクで発音する

    contentEl.innerHTML = `

      <p style="font-size:1.5rem; font-weight:bold;">マイクをおして、えいごでいってみよう！</p>

      <div class="question-emoji">${qData.emoji}</div>

      <div class="question-word">${qData.en}</div>

      <p style="font-size:1.2rem; color:#666;">（${qData.ja}）</p>

      <button class="mic-btn" id="mic-btn">🎤</button>

      <button id="play-voice-btn" style="margin-top:10px; width:60px; height:60px; font-size:1.5rem;">🔊 きく</button>

    `;

    document.getElementById('play-voice-btn').onclick = () => playVoice(qData.en);

    const micBtn = document.getElementById('mic-btn');

    

    micBtn.onclick = () => {

      micBtn.classList.add('listening');

      recognition.start();

    };

    

    recognition.onresult = (event) => {

      micBtn.classList.remove('listening');

      const spoken = event.results[0][0].transcript.toLowerCase().replace(/[^a-z0-9]/g, '');

      const target = qData.en.toLowerCase().replace(/[^a-z0-9]/g, '');

      

      // 5歳児向けに甘い判定（部分一致でOK）

      if (spoken.includes(target) || target.includes(spoken) || spoken.length > 0) {

        checkAnswer(true);

      } else {

        checkAnswer(false);

      }

    };

    recognition.onerror = () => { micBtn.classList.remove('listening'); checkAnswer(false); };

    

  } else {

    // 形式3：英語を見て正しい絵を選ぶ

    contentEl.innerHTML = `

      <p style="font-size:1.5rem; font-weight:bold;">えいごをよんで、えらぼう！</p>

      <div class="question-word" style="font-size:3.5rem; color:#E91E63;">${qData.en}</div>

      <div class="options-grid" id="options-container"></div>

    `;

    createOptions(qData, contentEl.querySelector('#options-container'), true);

  }

}


function createOptions(correctData, container, useEmoji) {

  // ダミーの選択肢を2つ選ぶ

  const dummies = [...wordData].filter(w => w.en !== correctData.en).sort(() => 0.5 - Math.random()).slice(0, 2);

  const options = [correctData, ...dummies].sort(() => 0.5 - Math.random());

  

  options.forEach(opt => {

    const btn = document.createElement('div');

    btn.className = 'option-btn';

    btn.innerHTML = `<span style="font-size:3rem;">${opt.emoji}</span> <span style="font-size:1.5rem;">${opt.ja}</span>`;

    btn.onclick = () => {

      checkAnswer(opt.en === correctData.en);

    };

    container.appendChild(btn);

  });

}


function playVoice(text) {

  const msg = new SpeechSynthesisUtterance(text);

  msg.lang = 'en-US';

  msg.rate = 0.8; // ゆっくり

  window.speechSynthesis.speak(msg);

}


function checkAnswer(isCorrect) {

  const overlay = document.getElementById('feedback-overlay');

  const icon = document.getElementById('feedback-icon');

  const text = document.getElementById('feedback-text');

  

  if (isCorrect) {

    icon.innerText = "⭕";

    text.innerText = "せいかい！すごい！";

    text.style.color = "var(--correct)";

    totalCorrect++;

  } else {

    icon.innerText = "❌";

    text.innerText = "おしい！もういちど！";

    text.style.color = "var(--incorrect)";

  }

  

  overlay.classList.add('show');

  

  // 効果音代わりに音声を再生

  const msg = new SpeechSynthesisUtterance(isCorrect ? "Good job!" : "Oh no");

  msg.lang = 'en-US';

  window.speechSynthesis.speak(msg);

  

  setTimeout(() => {

    overlay.classList.remove('show');

    if (isCorrect) {

      currentQIndex++;

      showNextQuestion();

    }

  }, 2000);

}
